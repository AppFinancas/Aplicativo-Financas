// Banco de Dados para App de Finanças com Open Finance

Table usuarios {
  id int [pk, increment]
  nome varchar(100) [not null]
  email varchar(255) [unique, not null]
  senha_hash varchar(255) [not null]
  telefone varchar(20)
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
}

Table conexoes_open_finance {
  id int [pk, increment]
  usuario_id int [not null]
  instituicao_nome varchar(100) [not null]
  instituicao_cnpj varchar(14) [not null]
  consent_id varchar(255) [unique, not null]
  consent_status varchar(50) [not null]
  consent_expiration_date timestamp
  link_id varchar(255)
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (usuario_id)
  }
}

Table contas {
  id int [pk, increment]
  conexao_id int [not null]
  account_id_externo varchar(255) [not null]
  tipo_conta varchar(50) [not null] // 'CONTA_DEPOSITO_A_VISTA', 'CONTA_POUPANCA', etc
  complemento_tipo varchar(50)
  numero_conta varchar(50)
  numero_agencia varchar(20)
  moeda varchar(3) [default: 'BRL']
  ultima_atualizacao timestamp [default: `now()`]

  indexes {
    (conexao_id, account_id_externo) [unique]
  }
}

Table cartoes_credito {
  id int [pk, increment]
  conexao_id int [not null]
  credit_card_account_id_externo varchar(255) [not null]
  nome varchar(100)
  bandeira varchar(50)

  indexes {
    (conexao_id, credit_card_account_id_externo) [unique]
  }
}

Table saldos_conta {
  id int [pk, increment]
  conta_id int [not null]
  data_saldo timestamp [not null]
  valor_disponivel decimal(15,2) [not null]
  valor_bloqueado decimal(15,2) [default: 0]
  valor_cheque_especial_utilizado decimal(15,2) [default: 0]
  data_referencia_api timestamp
  created_at timestamp [default: `now()`]

  indexes {
    (conta_id, data_saldo)
  }
}

Table limites_credito {
  id int [pk, increment]
  cartao_credito_id int [not null]
  data_limite timestamp [not null]
  limite_total decimal(15,2) [not null]
  limite_disponivel decimal(15,2) [not null]
  limite_utilizado decimal(15,2) [not null]
  data_referencia_api timestamp
  created_at timestamp [default: `now()`]

  indexes {
    (cartao_credito_id, data_limite)
  }
}

Table faturas_credito {
  id int [pk, increment]
  cartao_credito_id int [not null]
  bill_id_externo varchar(255)
  data_vencimento date [not null]
  data_fechamento date
  valor_total decimal(15,2) [not null]
  valor_pago decimal(15,2) [default: 0]
  status varchar(50) // 'ABERTA', 'FECHADA', 'VENCIDA', 'PAGA'
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (cartao_credito_id)
  }
}

Table categorias {
  id int [pk, increment]
  usuario_id int [not null]
  nome varchar(50) [not null]
  cor varchar(7) [default: '#CCCCCC']

  indexes {
    (usuario_id, nome) [unique]
  }
}

Table transacoes {
  id int [pk, increment]
  usuario_id int [not null]
  produto_id int [not null] // ID na tabela específica (contas ou cartoes_credito)
  tipo_produto tipo_produto_enum [not null] // 'conta' ou 'cartao_credito'
  transaction_id_externo varchar(255) [unique, not null]
  descricao text
  valor decimal(15,2) [not null]
  data_transacao date [not null]
  tipo_transacao varchar(100) // 'COMPRA_A_VISTA', 'PAGAMENTO', 'TED', etc
  direcao varchar(10) [not null] // 'entrada' ou 'saida'
  categoria_id int
  transferencia_id int
  created_at timestamp [default: `now()`]

  indexes {
    (usuario_id, data_transacao)
    (produto_id, tipo_produto)
    (categoria_id)
  }
}

Table objetivos {
  id int [pk, increment]
  usuario_id int [not null]
  nome varchar(100) [not null]
  descricao text
  data_alvo date
  valor_meta decimal(15,2)
  cor varchar(7) [default: '#4CAF50']
  status varchar(20) [default: 'ativo']
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]

  indexes {
    (usuario_id)
  }
}

Table alocacoes_objetivo {
  id int [pk, increment]
  objetivo_id int [not null]
  transacao_origem_id int
  valor_alocado decimal(15,2) [not null]
  data_alocacao timestamp [default: `now()`]
  observacao text

  indexes {
    (objetivo_id)
  }
}

// Enum para tipo_produto
Enum tipo_produto_enum {
  conta
  cartao_credito
}

// Relacionamentos

Ref: conexoes_open_finance.usuario_id > usuarios.id

Ref: contas.conexao_id > conexoes_open_finance.id

Ref: cartoes_credito.conexao_id > conexoes_open_finance.id

Ref: saldos_conta.conta_id > contas.id

Ref: limites_credito.cartao_credito_id > cartoes_credito.id

Ref: faturas_credito.cartao_credito_id > cartoes_credito.id

Ref: categorias.usuario_id > usuarios.id

Ref: transacoes.usuario_id > usuarios.id
Ref: transacoes.categoria_id > categorias.id
Ref: transacoes.transferencia_id > transacoes.id

 
Ref: objetivos.usuario_id > usuarios.id

Ref: alocacoes_objetivo.objetivo_id > objetivos.id
Ref: alocacoes_objetivo.transacao_origem_id > transacoes.id
